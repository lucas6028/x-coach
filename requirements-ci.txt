# Lean dependency set for CI (GitHub Actions).
#
# The full requirements.txt pins a heavy, platform-specific ML stack (torch, mediapipe,
# opencv, transformers) whose wheels are slow/fragile to install on a clean Linux runner.
# The backend and the dependency-light unit tests don't need that stack: the backend defers
# its ML imports and the tests mock them. This file installs only what those tests require,
# with loose, Linux-friendly version ranges.
#
# Heavy/data-dependent test modules are skipped in CI (see .github/workflows/ci.yml):
#   - test_mediapipe_smoothing / test_rtmpose_pose_extraction / test_videomae_video_classifier
#     / test_video_level_error_classification (need mediapipe / opencv / torch)
#   - test_backend_analysis (integration test against the real data/ tree, KG and RAG stores)

# Web backend + API test client
fastapi>=0.110,<1
python-multipart>=0.0.9
httpx>=0.27,<1
boto3>=1.34,<2   # Cloudflare R2 over the S3 API (backend/app/services/storage.py)

# tests/test_backend.py checks backend/app/auth.py against the REAL supabase-py auth error
# classes (AuthRetryableError -> 503, AuthApiError -> 401), not lookalikes, so the class-name
# match cannot silently rot. Only the auth sub-package is needed; the full `supabase` client
# stays out of the lean set (the backend defers that import and the tests fake it).
supabase_auth>=2.9,<3

pydantic-settings>=2.14.2,<3

# Reads the committed Lumen sprite sheet in tests/test_lumen_run_sheet.py, which checks that the
# PNG's frame count and square frames still agree with the CSS. That test has failed on `main`
# since it landed, because it is the first test to need an image library and this file was never
# told: PIL is present locally only as a transitive dependency of the heavy ML stack, which CI
# deliberately does not install. Pillow ships a Linux wheel, so the lean set stays lean.
pillow>=10,<13

# Imported at app load (src/knowledge) and by the light unit tests
numpy>=1.26,<3
networkx>=3,<4
langchain-text-splitters>=1.1.2,<2
pypdf>=4

# Test + coverage tooling
pytest>=8,<9
pytest-cov>=5,<7
coverage>=7,<8
